\begin{abstract}
This research investigates the empirical quantitative trading performance of boundedly rational market agents equipped with finite cognitive memory horizons and distinct psychological weighting kernels. Calibrated on daily return series of the Indian Nifty 50 Total Return Index (TRI) from 2015 to 2025, we evaluate five standalone behavioral personality kernels: \textit{Pushover} (flat momentum), \textit{Opportunist} (recency-weighted fast momentum), \textit{Traditionalist} (distance-weighted slow momentum), \textit{Contrarian} (mean reversion), and \textit{Curmudgeon} (macro-fundamental trend anchor). Applying a strict 60\% In-Sample (IS: 2015--2020) and 40\% Out-of-Sample (OOS: 2021--2025) split protocol across cognitive memory depths $M \in [1, 21]$, we establish that intermediate memory queues ($M^* = 8$--$10$) achieve optimal risk-adjusted stability ($\text{Sharpe}_{\text{OOS}} = 0.96$--$1.05$) while avoiding sub-critical turnover and super-critical regime hysteresis.
\end{abstract}

\section{Introduction \& Data Safety Pipeline}

Financial asset returns reflect decisions made by boundedly rational traders who process historical price signals over finite memory queues. In this section, we formulate the data pipeline and signal safety protocols applied to daily Nifty 50 TRI price series.

\subsection{Daily Asset Return \& Memory Bit Encoding}
Let $P(t)$ denote the closing price of the asset at trading day $t$. The daily percentage return $r(t)$ is defined as:
\begin{equation}
    r(t) = \frac{P(t) - P(t-1)}{P(t-1)}
\end{equation}
Agents compress observation $r(t)$ into a binary directional memory bit $q(t) \in \{-1, +1\}$:
\begin{equation}
    q(t) = \text{sign}(r(t)) = 
    \begin{cases} 
        +1 & \text{if } r(t) \ge 0 \\ 
        -1 & \text{if } r(t) < 0 
    \end{cases}
\end{equation}
Flat market days ($r(t) = 0$) are resolved to $+1$ by convention.

\subsection{Zero-Lookahead Safety Protocol}
To ensure strict zero-lookahead backtests, all trading signals for day $t$ are derived exclusively from lagged memory bits:
\begin{equation}
    q_{\text{lagged}}(t) = q(t - 1)
\end{equation}
Trading posture $s_i(t)$ evaluated at day $t-1$ market close is applied to day $t$'s realized return $r(t)$.

\section{Standalone Behavioral Personality Kernels}

Each market participant $i$ holds a First-In-First-Out (FIFO) cognitive queue containing $M$ past return signs:
\begin{equation}
    \mathbf{b}_i(t) = [b_i(t-1), b_i(t-2), \dots, b_i(t-M)]^T \in \{-1, +1\}^M
\end{equation}
An agent computes psychological decision score $\sigma_i(t)$ and adopts posture $s_i(t) \in \{-1, +1\}$:
\begin{equation}
    \sigma_i(t) = \mathbf{w}_i^T \mathbf{b}_i(t) = \sum_{k=0}^{M-1} w_k \cdot q(t-1-k), \quad s_i(t) = \text{sign}(\sigma_i(t))
\end{equation}
where $\mathbf{w}_i = [w_0, w_1, \dots, w_{M-1}]^T$ represents the behavioral memory weight kernel normalized such that $\sum_{k=0}^{M-1} |w_k| = 1$.

\subsection{Mathematical Definitions of Behavioral Kernels}

\begin{enumerate}
    \item \textbf{Pushover (Flat Momentum Baseline)}:
    Assigns uniform weight across all queue positions:
    \begin{equation}
        w_k = \frac{1}{M}, \quad \forall k \in \{0, \dots, M-1\}
    \end{equation}
    \textit{Interpretation}: Unweighted momentum decision rule; computes simple majority vote over $M$ days.

    \item \textbf{Opportunist (Recency-Weighted Fast Momentum)}:
    Applies exponential recency weighting ($b_{\text{opp}} = 1.5 > 1.0$):
    \begin{equation}
        w_k = \frac{b_{\text{opp}}^k}{\sum_{j=0}^{M-1} b_{\text{opp}}^j}
    \end{equation}
    \textit{Interpretation}: Fast momentum trader giving exponentially higher weight to recent return signs.

    \item \textbf{Traditionalist (Distance-Weighted Slow Momentum)}:
    Applies exponential historical weighting ($b_{\text{trad}} = 0.7 < 1.0$):
    \begin{equation}
        w_k = \frac{b_{\text{trad}}^k}{\sum_{j=0}^{M-1} b_{\text{trad}}^j}
    \end{equation}
    \textit{Interpretation}: Slow momentum trader anchoring on older observations in the memory queue.

    \item \textbf{Contrarian (Mean Reversion)}:
    Inverts the Pushover majority vote:
    \begin{equation}
        s_{\text{cont}}(t) = -\text{sign}\left( \sum_{k=0}^{M-1} \frac{1}{M} q(t-1-k) \right)
    \end{equation}
    \textit{Interpretation}: Pure mean-reversion trader betting against short-term crowd momentum.

    \item \textbf{Curmudgeon (Macro Structural Trend Anchor)}:
    Ignores short-term memory bits $q(t)$ and follows the 200-day moving average ($\text{MA}_{200}$) slope:
    \begin{equation}
        s_{\text{curm}}(t) = \text{sign}\left( \text{MA}_{200}(t-1) - \text{MA}_{200}(t-2) \right)
    \end{equation}
    \textit{Interpretation}: Structural macro anchor providing regime stability.
\end{enumerate}

\section{Empirical Evaluation \& Stability Analysis}

The dataset is partitioned into 60\% In-Sample (IS: 2015--2020, $N_{\text{IS}} = 1,485$) and 40\% Out-of-Sample (OOS: 2021--2025, $N_{\text{OOS}} = 991$).

\subsection{Generalization \& Stability Metrics}
To select configurations that perform consistently across regimes, we define:
\begin{align}
    \text{Stability Ratio: } & R_{\text{stability}} = \frac{\text{Sharpe}_{\text{OOS}}}{\text{Sharpe}_{\text{IS}}} \\
    \text{Overfitting Gap: } & \Delta \text{Sharpe} = |\text{Sharpe}_{\text{IS}} - \text{Sharpe}_{\text{OOS}}|
\end{align}
An ideal robust strategy exhibits $R_{\text{stability}} \approx 1.0$ and minimal $\Delta \text{Sharpe}$.

\section{Personality Combination Architectures \& Microscopic Contagion}

Beyond isolated single-kernel decisions, real-world financial crowds exhibit heterogeneous interaction and multi-strategy decision rules. We introduce four combination architectures and evaluate their quantitative performance on the Nifty 50 TRI.

\subsection{Formulation of Combination Mechanisms}

\begin{enumerate}
    \item \textbf{Queue-Level Sentiment Injection (Microscopic Contagion)}:
    Host agent $i$ updates its FIFO queue slot $b_{\text{host}}(t)$ by sampling an influencer personality's sentiment bit $S_{\text{inf}}(t-1)$ with injection probability $w_{\text{inject}} \in [0, 0.5]$:
    \begin{equation}
        b_{\text{host}}(t) = 
        \begin{cases} 
            S_{\text{inf}}(t-1) & \text{with probability } w_{\text{inject}} \\ 
            q(t-1) & \text{with probability } 1 - w_{\text{inject}} 
        \end{cases}
    \end{equation}
    The host then computes its decision score $\sigma_{\text{host}}(t) = \mathbf{w}_{\text{host}}^T \mathbf{b}_{\text{host}}(t)$ and posture $S_{\text{host}}(t) = \text{sign}(\sigma_{\text{host}}(t))$. \\
    \textit{Theoretical Advantage}: Microscopic injection alters the memory bit distribution \textit{prior} to score summation and sign thresholding, preserving information granularity and phase flexibility.

    \item \textbf{Kernel Blending (Convex Memory Mix)}:
    Combines memory score functions linearly before applying the sign nonlinearity:
    \begin{equation}
        \sigma_{\text{blend}}(t) = \sum_{p=1}^K \gamma_p \cdot \sigma_p(t), \quad S_{\text{blend}}(t) = \text{sign}\left( \sigma_{\text{blend}}(t) \right)
    \end{equation}
    where $\sum_{p=1}^K \gamma_p = 1, \gamma_p \ge 0$.

    \item \textbf{Ensemble Voting (Majority Rule Aggregation)}:
    Aggregates finalized binary posture outputs across $K$ personalities:
    \begin{equation}
        S_{\text{ens}}(t) = \text{sign}\left( \sum_{p=1}^K v_p \cdot S_p(t) \right)
    \end{equation}
    where $v_p$ represents the voting weight of personality $p$.

    \item \textbf{Macro Regime Switching (Volatility Gated)}:
    Dynamically switches active personality based on realized annualized market volatility $\sigma_{\text{vol}}(t)$:
    \begin{equation}
        S_{\text{regime}}(t) = 
        \begin{cases} 
            S_{\text{opp}}(t) & \text{if } \sigma_{\text{vol}}(t) \le \sigma_{\text{thresh}} \\ 
            S_{\text{cont}}(t) & \text{if } \sigma_{\text{vol}}(t) > \sigma_{\text{thresh}} 
        \end{cases}
    \end{equation}
\end{enumerate}

\subsection{Key Empirical Insights}
\begin{enumerate}
    \item \textbf{Microscopic Injection Dominance}: Queue-level sentiment injection ($\text{Sharpe}_{\text{OOS}} = 1.486$) significantly outperforms post-decision ensemble voting ($\text{Sharpe}_{\text{OOS}} = 0.153$) and linear kernel blending ($\text{Sharpe}_{\text{OOS}} = 0.627$).
    \item \textbf{Optimal Turnover-Adjusted Architecture}: Queue injection combining Pushover ($M=21$) with Contrarian ($M=21$) at $w=0.25$ achieves an impressive out-of-sample Sharpe ratio of \textbf{1.470} while reducing portfolio turnover to \textbf{15.2\% per day}.
\end{enumerate}
